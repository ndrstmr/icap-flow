23/**
24 * A bound socket that can carry more than one write/read round-trip
25 * before being released.
26 *
27 * The strict RFC 3507 §4.5 preview-continue flow needs this: the
28 * client sends the preview, reads the server's `100 Continue`, then
29 * sends the rest of the body — all on the same connection, all part
30 * of one logical ICAP request. The plain
31 * {@see TransportInterface::request()} method is one-shot and can't
32 * model that.
33 *
34 * Sessions are obtained from
35 * {@see SessionAwareTransport::openSession()}. Callers MUST call
36 * either {@see release()} (return to the pool when configured, close
37 * otherwise) or {@see close()} (force-close), exactly once.
38 */
39interface TransportSession
40{
41    /**
42     * Push request bytes onto the socket, all chunks of the call within one
43     * window of `streamTimeout` seconds and under the cancellation token the
44     * session was opened with. A call that does not finish ends the session
45     * for good (it is closed, and {@see release()} cannot pool it): a
46     * cancellation by the caller, also a `TimeoutCancellation`, throws
47     * `Amp\CancelledException`, the window passing throws
48     * {@see \Ndrstmr\Icap\Exception\IcapTimeoutException}. Both are looked
49     * at whenever the producer hands a chunk back, an empty one too, and
50     * before the call returns: a producer that is slow, or blocks the event
51     * loop, cannot extend the window or finish the call after it.
52     *
53     * @param iterable<string> $chunks
54     */
55    public function write(iterable $chunks): void;
56
57    /**
58     * Read one fully-framed ICAP response from the socket. The response ends
59     * where its layout says ({@see ResponseFrameReader}); one that is illegal,
60     * or that the stream cuts off, throws
61     * {@see \Ndrstmr\Icap\Exception\IcapMalformedResponseException} and ends
62     * the session. A read can bring in more than the response: what lies behind
63     * its end stays with the session and is the start of what the next call
64     * returns. No request waits for such bytes, so a {@see write()} on top of
65     * them throws and closes the session, and {@see release()} closes the
66     * socket instead of pooling it. Cancellation and failure behave as for
67     * {@see write()}; the windows are two, one after the other (M10): the
68     * server's time to begin its answer once the request is sent
69     * (`Config::getResponseStartTimeout()`), then, from the first byte, a
70     * maximum for the answer to arrive in full (`Config::getStreamTimeout()`)
71     * that no further byte starts again.
72     */
73    public function readResponse(): string;
74
75    /**
76     * Return the socket to the underlying pool when one is configured;
77     * otherwise close it. Call it only after the whole exchange has been
78     * validated (frame, status, verdict, no `Connection: close`): a
79     * released socket is out of reach of {@see close()}. A socket that holds
80     * bytes behind the last response is closed instead. Idempotent in
81     * the sense that further calls are no-ops.
82     */
83    public function release(): void;

89     * request flow).
90     */
91    public function close(): void;
