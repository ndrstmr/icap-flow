34/**
35 * Asynchronous transport implementation using amphp/socket.
36 *
37 * Upgrades to TLS (icaps://) automatically when the supplied Config
38 * carries a {@see \Amp\Socket\ClientTlsContext}; otherwise connects
39 * plain tcp://. The response is bounded by Config::maxResponseSize to
40 * keep a hostile server from exhausting the client's memory.
41 *
42 * The optional user-supplied {@see Cancellation} is combined with the
43 * transport's own `streamTimeout` window for each phase (acquiring the
44 * socket, writing the request, reading the response); whichever comes
45 * first ends the phase. The caller's cancellation — also one that is a
46 * `TimeoutCancellation` — ends it with `Amp\CancelledException`, the
47 * transport's own window with {@see IcapTimeoutException}.
48 *
49 * Response framing is done by {@see ResponseFrameReader} so the read
50 * loop terminates as soon as the message is complete — no dependency
51 * on the server closing the socket.
52 *
53 * **Connection pooling (v2.1+).** When constructed with a
54 * {@see ConnectionPoolInterface}, the transport calls `acquire()`
55 * instead of opening a fresh TCP/TLS connection. A socket goes back
56 * with `release()` only when the caller of a {@see openSession()}
57 * session has validated the whole exchange; every other way out closes
58 * it. {@see request()} returns the raw response string, before any
59 * status or verdict has been checked, so it always closes the socket.
60 * Without a pool the transport opens a fresh socket per request and
61 * closes it after — identical to the v2.0 behaviour. A pool that limits
62 * its connections ({@see AmpConnectionPool}) limits the sessions and
63 * requests of this transport too: a session is a connection, and its room
64 * is back when the socket is closed or released. Without a pool the
65 * application limits its own concurrency.
66 *
67 * **Sessions (v2.1+).** Implements {@see SessionAwareTransport} so
68 * callers that need to send + receive multiple times on the same
69 * connection (strict RFC 3507 §4.5 preview-continue, ICAP request
70 * pipelining, …) can call {@see openSession()}.
71 */

135        $cancellation?->throwIfRequested();

137        // Waiting for the pool, connecting and the TLS handshake share one window: streamTimeout seconds, or the
138        // handshake limit if that is longer (the window must not cut it short). The end of the whole operation reaches
139        // the window as the cancellation of the caller and is shorter if it is. Timer and subscription are removed on
