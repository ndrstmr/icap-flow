31/**
32 * One TCP connection of the {@see SynchronousStreamTransport}, for as many write/read round trips as an exchange
33 * needs: the preview and the rest of the body of RFC 3507 §4.5 travel on it as one request. Calls block, and there is
34 * no TLS; use {@see AsyncAmpTransport} for icaps://.
35 *
36 * A request goes out through a non-blocking socket that is waited for with stream_select(): a blocking send() waits for
37 * ever on macOS for a peer that has stopped reading. A chunk is sent completely or the call fails. A {@see write()} has
38 * one window of `streamTimeout` for all its chunks, the time its producer takes included, as in
39 * {@see AmpTransportSession}; the request of {@see SynchronousStreamTransport::request()}, which has the connection to
40 * itself, keeps the limit of that transport ({@see sendRequest()}). A response is read in blocking mode in the two
41 * windows of {@see AmpTransportSession}: `responseStartTimeout` for the server to begin, then `streamTimeout` from its
42 * first byte as a maximum for the response to arrive in full.
43 *
44 * A native wait cannot be interrupted by the caller's cancellation, which in an operation of the client is also its end
45 * ({@see OperationDeadline}). Each wait is bounded by the time that is left of the operation and of its own limit, and
46 * both are looked at between the waits: before the producer is asked for a first chunk, whenever it hands one back,
47 * after every write and every wait before the result counts, and after every read before its bytes, its end of the
48 * stream or its failure are told. A producer that does not return, or a call that PHP cannot interrupt, is noticed when
49 * control comes back. A call that does not finish closes the session for good; a limit that passes is an
50 * {@see IcapTimeoutException}, a cancellation of the caller stays what it is and comes first.
51 *
52 * A read that brings in bytes behind the end of a response fails, as in {@see AmpTransportSession}: they answer no
53 * request.
54 */

349     * call waits for the peer to read: until the end of the window of the
350     * call ($windowEndsAt, hrtime(true)), or without a window for one timeout

359        $length = strlen($chunk);
360        $offset = 0;
361        $endsAt = $windowEndsAt;
362        while (true) {
363            // PHP raises a notice for every send that fails; the failure is reported below.
364            $written = @fwrite(
365                $stream,
366                $offset === 0 && $length <= self::WRITE_SLICE_SIZE
367                    ? $chunk
368                    : substr($chunk, $offset, self::WRITE_SLICE_SIZE),
369            );
370            // What the write did, a failure too, counts only if the caller has not cancelled and neither the operation
371            // nor the window ran out meanwhile.
372            self::lookAt($cancellation, $windowEndsAt, $timeout);
373            if ($written === false) {
374                throw new IcapConnectionException('Writing the request failed.');
375            }
376
377            $offset += $written;
378            if ($offset >= $length) {
379                return;
380            }
381
382            if ($written > 0) {
383                // Progress starts the time without progress again, not the window.
384                $endsAt = $windowEndsAt;
385            } else {
386                $endsAt ??= hrtime(true) + $stall;
387                self::awaitWritable($stream, $endsAt, $timeout, $cancellation);
388                // The rest of the chunk goes out only if the caller still wants it, and the operation and the window
389                // still have time.
390                self::lookAt($cancellation, $windowEndsAt, $timeout);
391            }
392        }
393    }

395    /**
396     * Waits until the socket may take more, or throws once $endsAt has passed.
397     * The wait is no longer than the time that is left of the operation. A
398     * wait that stream_select() cannot do (a signal interrupted it, or the
399     * descriptor lies beyond FD_SETSIZE) is no failure of the request: it
400     * returns after a moment and the caller tries to write again.
401     *
402     * @param resource $stream
403     * @param float    $endsAt hrtime(true) at which the wait is over
404     */
405    private static function awaitWritable($stream, float $endsAt, float $timeout, ?Cancellation $cancellation = null): void
406    {
407        $left = ($endsAt - hrtime(true)) / 1e9;
408        if ($cancellation instanceof OperationDeadline) {
409            $left = min($left, $cancellation->remaining());
410        }
411        if ($left <= 0.0) {
412            // The end of the operation, if that is what ran out, is told as such.
413            self::look($cancellation);
414            throw self::timeout('Writing the request', $timeout);
415        }
416
417        $read = null;
418        $write = [$stream];
419        $except = null;
420        [$seconds, $microseconds] = self::splitTimeout($left);
421        if (@stream_select($read, $write, $except, $seconds, $microseconds) === false) {
422            usleep(1_000);
423        }
424    }

426    /**
427     * One read of the response, null at the end of the stream. A read that
428     * timed out is no end of the stream (the frame reader would call it a
429     * connection closed too early) but a timeout. PHP reports one as false
430     * with `timed_out` set; a zero timeout answers with an empty string while
431     * the stream has not ended.
432     *
433     * With $phaseEndsAt (hrtime(true) at which the phase is over) the read waits for what is left of the phase,
434     * not for a whole timeout: a maximum, which no byte starts again. It waits no longer than the time that is
435     * left of the operation either.
436     *
437     * @param resource $stream
438     */
439    private static function readChunk($stream, ?Cancellation $cancellation, float $timeout, ?float $phaseEndsAt = null, string $what = 'Reading the response'): ?string
440    {
441        self::look($cancellation);
442
443        if ($phaseEndsAt !== null) {
444            $left = ($phaseEndsAt - hrtime(true)) / 1e9;
445            if ($cancellation instanceof OperationDeadline) {
446                $left = min($left, $cancellation->remaining());
447            }
448            if ($left <= 0.0) {
449                self::look($cancellation);
450                throw self::timeout($what, $timeout);
451            }
452            [$seconds, $microseconds] = self::splitTimeout($left);
453            stream_set_timeout($stream, $seconds, $microseconds);
454        }
455
456        $read = fread($stream, self::READ_CHUNK_SIZE);
457
458        // What the read brought - bytes, the end of the stream, a failure - counts only if the caller did not cancel and
459        // neither the operation nor the phase ran out while it blocked: a peer that hung up after that is no framing
460        // error, and bytes that came late are no answer.
461        self::look($cancellation);
462        if ($phaseEndsAt !== null && hrtime(true) >= $phaseEndsAt) {
463            throw self::timeout($what, $timeout);
464        }
465
466        if ($read !== false && $read !== '') {
467            return $read;
468        }
469
470        $meta = stream_get_meta_data($stream);
471        if ($meta['timed_out'] || ($read === '' && !$meta['eof'])) {
472            throw self::timeout($what, $timeout);
473        }
474        if ($read === false) {
475            throw new IcapConnectionException('Reading the response failed.');
476        }
477
478        return null;
479    }
