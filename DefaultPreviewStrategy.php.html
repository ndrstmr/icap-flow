28/**
29 * Default strategy for the answer to a preview (RFC 3507 §4.5).
30 *
31 * A server answers a preview with 100 Continue (send the rest of the body),
32 * 204 No Content (no change) or a final response, which may come before the
33 * rest of the body was sent. This class applies the standard rules, which
34 * are verified against no ICAP product: 204 counts as clean, and a 200/206
35 * is infected when it carries one of the configured virus headers. The
36 * verified mode ({@see CicapVerdict}) does not ask this class.
37 */
38final class DefaultPreviewStrategy implements PreviewStrategyInterface
39{
40    /** @var list<string> */
41    private array $virusFoundHeaders;
42
43    /**
44     * @param list<string> $virusFoundHeaders Ordered list of vendor virus-name
45     *   headers to inspect when a 200/206 preview response arrives.  The first
46     *   header present in the response wins.  Defaults to ['X-Virus-Name'] for
47     *   backward compatibility; production deployments should pass
48     *   Config::getVirusFoundHeaders().
49     */
50    public function __construct(array $virusFoundHeaders = ['X-Virus-Name'])
51    {
52        $this->virusFoundHeaders = $virusFoundHeaders;
53    }
54
55    /**
56     * Decide whether to continue sending the body after a preview response.
57     *
58     * Status semantics:
59     * - 100 Continue   → server wants the rest of the body.
60     * - 204 No Content → server changes nothing; clean under the standard rules.
61     * - 200 / 206      → final response; inspect virus headers to
62     *                    determine whether the file is infected or clean.
63     * - 4xx            → {@see IcapClientException}, 5xx → {@see IcapServerException}:
64     *                    the same types as for the answer to any other request.
65     * - anything else  → protocol error.
66     *
67     * @throws IcapClientException   When the status is in the 4xx range
68     * @throws IcapServerException   When the status is in the 5xx range
69     * @throws IcapProtocolException When the status is none of the above
70     */
71    #[\Override]

73    {

81            default => throw IcapStatus::failure($code) ?? new IcapProtocolException(
82                'Unexpected preview status code: ' . $code,
83                $code,
84            ),
85        };
86    }
87
88    /**
