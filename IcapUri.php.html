24 * Builds and checks the URIs the client sends and connects to
25 * (RFC 3507 §4.2 with the RFC 3986 syntax). Host, port, service path and

29 * @internal
30 */

37    private const string QUERY = '(?:\?(?:' . self::PCHAR . '|[/?])*)?';
